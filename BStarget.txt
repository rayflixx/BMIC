
#include <iostream>
using namespace std;

int main() {
    int n, target;
    int a[100];

    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter sorted elements: ";
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    cout << "Enter target element: ";
    cin >> target;

    int low = 0, high = n - 1, mid;
    int index = -1;

    while (low <= high) {
        mid = low + (high - low) / 2;

        if (a[mid] == target) {
            index = mid;
            break;
        }
        else if (a[mid] < target) {
            low = mid + 1;
        }
        else {
            high = mid - 1;
        }
    }

    cout << index << endl;

    return 0;
}
