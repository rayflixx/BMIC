#include <iostream>
using namespace std;

struct Employee {
    char username[30];
    unsigned long long hashedPassword;
    int occupied;
};

Employee table[10];

// Simple password hashing function for demonstration
unsigned long long hashPassword(const char password[]) {
    unsigned long long hash = 5381;

    for (int i = 0; password[i] != '\0'; i++) {
        hash = hash * 33 + password[i];
    }

    return hash;
}

int hashUsername(const char username[]) {
    int hash = 0;

    for (int i = 0; username[i] != '\0'; i++) {
        hash = (hash * 31 + username[i]) % 10;
    }

    return hash;
}

void registerEmployee() {
    char username[30], password[30];

    cout << "Enter username: ";
    cin >> username;

    cout << "Enter password: ";
    cin >> password;

    int index = hashUsername(username);

    for (int i = 0; i < 10; i++) {
        int pos = (index + i) % 10;

        if (table[pos].occupied &&
            __builtin_strcmp(table[pos].username, username) == 0) {
            cout << "Username already exists!\n";
            return;
        }

        if (!table[pos].occupied) {
            int j = 0;
            while (username[j] != '\0') {
                table[pos].username[j] = username[j];
                j++;
            }
            table[pos].username[j] = '\0';

            table[pos].hashedPassword = hashPassword(password);
            table[pos].occupied = 1;

            cout << "Employee registered successfully!\n";
            return;
        }
    }

    cout << "Hash table is full!\n";
}

void authenticate() {
    char username[30], password[30];

    cout << "Enter username: ";
    cin >> username;

    cout << "Enter password: ";
    cin >> password;

    int index = hashUsername(username);

    for (int i = 0; i < 10; i++) {
        int pos = (index + i) % 10;

        if (!table[pos].occupied) {
            break;
        }

        if (__builtin_strcmp(table[pos].username, username) == 0) {
            if (table[pos].hashedPassword == hashPassword(password))
                cout << "Authentication successful!\n";
            else
                cout << "Incorrect password!\n";

            return;
        }
    }

    cout << "Employee not found!\n";
}

int main() {
    for (int i = 0; i < 10; i++)
        table[i].occupied = 0;

    int choice;

    do {
        cout << "\n1. Register Employee";
        cout << "\n2. Authenticate Employee";
        cout << "\n3. Exit";
        cout << "\nEnter choice: ";
        cin >> choice;

        switch (choice) {
            case 1:
                registerEmployee();
                break;
            case 2:
                authenticate();
                break;
            case 3:
                cout << "Exiting...\n";
                break;
            default:
                cout << "Invalid choice!\n";
        }
    } while (choice != 3);

    return 0;
}
